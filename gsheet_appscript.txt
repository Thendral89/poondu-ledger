// 🛡️ Production Spreadsheet Connection
const SPREADSHEET_ID = '1LUGAEclAgBU2H3KJhYPBlm8mTlIKy32gTA_b0z2ywlc';

// 🛡️ Master Schema: The Enforcer uses this to perfectly rebuild your sheet and wipe duplicates
const SHEET_HEADERS = {
  'lots': ['ID', 'Label', 'Bags', 'Bags G0', 'Bags G1', 'KG', 'Cost', 'Loading Cost', 'Grading Cost', 'Source of Funds', 'Weight Loss KG', 'Date Added', 'Location', 'Added By', 'Deleted Flag'],
  'sales': ['ID', 'Date', 'Lot ID', 'Channel', 'Contact ID', 'Bags', 'KG', 'Amount', 'Note', 'Added By', 'Deleted Flag'],
  'expenses': ['ID', 'Date', 'Type', 'Description', 'Amount', 'Reference ID', 'Bags', 'Bags G0', 'Bags G1', 'KG', 'Added By', 'Deleted Flag'],
  'capitalLoans': ['ID', 'Partner', 'Amount', 'Date Sent', 'Return Date', 'Note', 'Added By', 'Deleted Flag'],
  'contacts': ['ID', 'Name', 'Type', 'Phone', 'Location', 'Note', 'Deleted Flag']
};

const FIELD_MAP = {
  'ID': 'id', 'Deleted Flag': 'isDeleted', 'Added By': 'addedBy', 'Date': 'date', 'Note': 'note',
  'Amount': 'amount', 'Bags': 'bags', 'KG': 'kg', 'Bags G0': 'bagsG0', 'Bags G1': 'bagsG1',
  'Label': 'label', 'Cost': 'cost', 'Loading Cost': 'loadingCost', 'Grading Cost': 'gradingCost',
  'Source of Funds': 'sourceOfFunds', 'Weight Loss KG': 'weightLossKg', 'Date Added': 'dateAdded',
  'Location': 'location', 'Lot ID': 'lotId', 'Channel': 'channel', 'Contact ID': 'contactId',
  'Type': 'type', 'Description': 'desc', 'Reference ID': 'referenceId', 'Partner': 'partner',
  'Date Sent': 'dateSent', 'Return Date': 'returnDate', 'Name': 'name', 'Phone': 'phone'
};

const DATE_FIELDS = ['Date', 'Date Added', 'Date Sent', 'Return Date'];

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const action = e.parameter.action;
    
    if (action) {
        const payload = e.parameter.payload ? JSON.parse(e.parameter.payload) : null;
        return outputJson_(processAction_(ss, action, payload, e.parameter.id), e.parameter.callback);
    } else {
        const data = {
            lots: getObjects_(ss, 'lots'),
            sales: getObjects_(ss, 'sales'),
            expenses: getObjects_(ss, 'expenses'),
            capitalLoans: getObjects_(ss, 'capitalLoans'),
            contacts: getObjects_(ss, 'contacts'),
            partners: getPartners_(ss),
            config: getConfig_(ss),
            status: 'success'
        };
        return outputJson_(data, e.parameter.callback);
    }
  } catch(error) {
    return outputJson_({status: 'error', message: error.toString()}, e.parameter ? e.parameter.callback : null);
  }
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let action = '', payload = null, id = '';

    if (e.parameter.action) {
      action = e.parameter.action;
      payload = e.parameter.payload ? JSON.parse(e.parameter.payload) : null;
      id = e.parameter.id;
    } else if (e.postData && e.postData.contents) {
      const body = JSON.parse(e.postData.contents);
      action = body.action; payload = body.payload; id = body.id;
    }

    if (action) return outputJson_(processAction_(ss, action, payload, id));
    return outputJson_({ status: 'error', message: 'No action specified in POST request.' });
  } catch(error) {
    return outputJson_({ status: 'error', message: error.toString() });
  }
}

function outputJson_(data, callback) {
  const out = JSON.stringify(data);
  if (callback) return ContentService.createTextOutput(callback + '(' + out + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

function enforceStrictSchema_(ss, sheetName) {
  const requiredHeaders = SHEET_HEADERS[sheetName];
  let sh = ss.getSheetByName(sheetName);
  
  if (!sh) {
    sh = ss.insertSheet(sheetName);
    sh.appendRow(requiredHeaders);
    sh.getRange(1, 1, 1, requiredHeaders.length).setFontWeight("bold").setBackground("#EDE6D6");
    return requiredHeaders;
  }
  
  const data = sh.getDataRange().getValues();
  if (data.length === 0) {
    sh.appendRow(requiredHeaders);
    sh.getRange(1, 1, 1, requiredHeaders.length).setFontWeight("bold").setBackground("#EDE6D6");
    return requiredHeaders;
  }
  
  const currentHeaders = data[0].map(h => String(h).trim());
  let needsRebuild = false;
  
  // Check if rebuilding is necessary
  if (currentHeaders.length !== requiredHeaders.length) {
    needsRebuild = true;
  } else {
    for (let i = 0; i < requiredHeaders.length; i++) {
      if (currentHeaders[i] !== requiredHeaders[i]) { needsRebuild = true; break; }
    }
  }
  
  if (!needsRebuild) return requiredHeaders;
  
  // Rebuild sheet to strict headers, resolving any duplicates by taking the right-most valid data
  const newRows = [requiredHeaders];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (row.join('').trim() === '') continue; // Skip totally blank rows
    
    const newRow = requiredHeaders.map(reqH => {
      // Find all indices where this header exists in the current messy sheet
      let indices = [];
      for (let c = 0; c < currentHeaders.length; c++) {
        if (currentHeaders[c] === reqH) indices.push(c);
      }
      
      if (indices.length === 0) return '';
      
      // If duplicate columns exist, take the right-most one that has actual data
      let val = '';
      for (let idx of indices.reverse()) {
        if (row[idx] !== '' && row[idx] !== undefined) {
          val = row[idx];
          break;
        }
      }
      return val;
    });
    newRows.push(newRow);
  }
  
  sh.clear(); // Wipe the messy duplicated data
  sh.getRange(1, 1, newRows.length, requiredHeaders.length).setValues(newRows);
  sh.getRange(1, 1, 1, requiredHeaders.length).setFontWeight("bold").setBackground("#EDE6D6");
  
  return requiredHeaders;
}

function cleanDate_(val) {
  if (!val) return '';
  if (val instanceof Date) {
    const offset = 5.5 * 60 * 60 * 1000;
    return new Date(val.getTime() + offset).toISOString().slice(0, 10);
  }
  const str = String(val).trim();
  if (/^\d{4}-\d{2}-\d{2}$/.test(str)) return str;
  const parsed = new Date(str);
  if (!isNaN(parsed.getTime())) {
     const offset = 5.5 * 60 * 60 * 1000;
     return new Date(parsed.getTime() + offset).toISOString().slice(0, 10);
  }
  return str.slice(0, 10);
}

function getObjects_(ss, sheetName) {
  const headers = enforceStrictSchema_(ss, sheetName);
  const sh = ss.getSheetByName(sheetName);
  if (sh.getLastRow() < 2) return [];
  
  const values = sh.getDataRange().getValues();
  const deletedIndex = headers.indexOf('Deleted Flag');

  return values.slice(1)
    .filter(row => row.join('').trim() !== '')
    .filter(row => {
       // Hide soft-deleted rows from the UI
       if (deletedIndex === -1) return true;
       const delFlag = String(row[deletedIndex]).toUpperCase();
       return delFlag !== 'TRUE' && delFlag !== 'YES';
    })
    .map(row => {
      let obj = {};
      headers.forEach((h, i) => {
         const key = FIELD_MAP[h] || h; 
         if (DATE_FIELDS.includes(h)) obj[key] = cleanDate_(row[i]);
         else obj[key] = row[i];
      });
      obj.isDeleted = false; // Set to false in memory so frontend works normally
      return obj;
    });
}

function deleteRowById_(ss, sheetName, id) {
  const headers = enforceStrictSchema_(ss, sheetName);
  const sh = ss.getSheetByName(sheetName);
  if (sh.getLastRow() < 2) return { status: 'error', message: 'Sheet empty.' };
  
  const values = sh.getDataRange().getValues();
  const idIndex = headers.indexOf('ID');
  const deletedIndex = headers.indexOf('Deleted Flag');
  
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][idIndex]) === String(id)) {
      if (deletedIndex !== -1) sh.getRange(i + 1, deletedIndex + 1).setValue('TRUE'); // Soft Delete
      else sh.getRange(i + 1, headers.length).setValue('TRUE'); 
      return { status: 'success', deleted: true };
    }
  }
  return { status: 'error', message: 'Delete failed: ID not found.' };
}

function addOrUpdateRow_(ss, sheetName, payload, isNew, id) {
  const headers = enforceStrictSchema_(ss, sheetName);
  const sh = ss.getSheetByName(sheetName);
  
  const rowData = headers.map(h => {
      const key = FIELD_MAP[h] || h;
      if (h === 'Deleted Flag') return payload[key] === true || String(payload[key]).toUpperCase() === 'TRUE' ? 'TRUE' : 'FALSE';
      return payload[key] !== undefined ? payload[key] : '';
  });

  if (isNew) {
      sh.appendRow(rowData);
      return { status: 'success', updated: true };
  } else {
      const values = sh.getDataRange().getValues();
      const idIndex = headers.indexOf('ID');
      for (let i = 1; i < values.length; i++) {
          if (String(values[i][idIndex]) === String(id)) {
              sh.getRange(i + 1, 1, 1, rowData.length).setValues([rowData]);
              return { status: 'success', updated: true };
          }
      }
  }
  return { status: 'error', message: 'Update failed: Record ID not found.' };
}

function processAction_(ss, action, payload, id) {
  const entityMap = { 'Lot': 'lots', 'Sale': 'sales', 'Expense': 'expenses', 'CapitalLoan': 'capitalLoans', 'Contact': 'contacts' };

  for (let entity in entityMap) {
      if (action === `add${entity}`) return addOrUpdateRow_(ss, entityMap[entity], payload, true);
      if (action === `update${entity}`) return addOrUpdateRow_(ss, entityMap[entity], payload, false, id);
      if (action === `delete${entity}`) return deleteRowById_(ss, entityMap[entity], id);
  }

  if (action === 'updateConfig') { updateConfig_(ss, payload); return {status:'success'}; }
  if (action === 'updatePartners') { updatePartners_(ss, payload); return {status:'success'}; }
  return { status: 'error', message: 'Unknown action requested.' };
}

function getConfig_(ss) {
  const sh = ss.getSheetByName('config');
  if (!sh) return { rent1: 45000, rent2: 10000, labour: 95000, interestRatePct: 1.5 };
  const data = sh.getDataRange().getValues();
  let cfg = {};
  for(let i=1; i<data.length; i++) cfg[data[i][0]] = data[i][1];
  return cfg;
}

function updateConfig_(ss, payload) {
  let sh = ss.getSheetByName('config');
  if (!sh) sh = ss.insertSheet('config');
  sh.clear();
  const rows = [['Key', 'Value']];
  for (let k in payload) rows.push([k, payload[k]]);
  sh.getRange(1, 1, rows.length, 2).setValues(rows);
}

function getPartners_(ss) {
  const sh = ss.getSheetByName('partners');
  if (!sh) return {
     Thendral: {required:650000, contributed:1100000},
     Nagaraj:  {required:650000, contributed:289550},
     Senthil:  {required:650000, contributed:625953}
  };
  const data = sh.getDataRange().getValues();
  let p = {};
  for(let i=1; i<data.length; i++) p[data[i][0]] = { required: Number(data[i][1])||0, contributed: Number(data[i][2])||0 };
  return p;
}

function updatePartners_(ss, payload) {
  let sh = ss.getSheetByName('partners');
  if (!sh) sh = ss.insertSheet('partners');
  sh.clear();
  const rows = [['Name', 'Required', 'Contributed']];
  for (let k in payload) rows.push([k, payload[k].required, payload[k].contributed]);
  sh.getRange(1, 1, rows.length, 3).setValues(rows);
}
